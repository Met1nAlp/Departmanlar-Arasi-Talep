#include "eventlogger.h"

EventLogger::EventLogger() {
    initTable();
}

bool EventLogger::initTable() {
    QSqlQuery query;
    bool success = query.exec(
        "CREATE TABLE IF NOT EXISTS request_events ("
        "id INTEGER PRIMARY KEY AUTOINCREMENT, "
        "request_id TEXT NOT NULL, "
        "old_status TEXT, "
        "new_status TEXT NOT NULL, "
        "occurred_at TEXT NOT NULL)"
    );

    if (!success) {
        qDebug() << "[EVENT_LOGGER HATA] Tablo olusturulamadi:" << query.lastError().text();
    }
    return success;
}

bool EventLogger::logRequestEvent(const QString &requestId, const QString &oldStatus, const QString &newStatus) {
    QSqlQuery query;
    query.prepare(
        "INSERT INTO request_events (request_id, old_status, new_status, occurred_at) "
        "VALUES (:reqId, :oldStatus, :newStatus, :occurredAt)"
    );

    QString nowUtc = QDateTime::currentDateTimeUtc().toString(Qt::ISODateWithMs);

    query.bindValue(":reqId", requestId.trimmed());
    query.bindValue(":oldStatus", oldStatus.trimmed());
    query.bindValue(":newStatus", newStatus.trimmed());
    query.bindValue(":occurredAt", nowUtc);

    if (query.exec()) {
        qDebug() << "[EVENT_LOGGER] Olay kaydedildi -> ID:" << requestId
                 << "|" << oldStatus << "->" << newStatus;
        return true;
    }

    qDebug() << "[EVENT_LOGGER HATA] Olay kaydedilemedi:" << query.lastError().text();
    return false;
}

QJsonArray EventLogger::getEventsSince(const QString &sinceTimestamp) {
    QJsonArray eventsArray;
    QSqlQuery query;

    query.prepare(
        "SELECT request_id, old_status, new_status, occurred_at "
        "FROM request_events "
        "WHERE occurred_at > :since "
        "ORDER BY occurred_at ASC"
    );
    query.bindValue(":since", sinceTimestamp.trimmed());

    if (query.exec()) {
        while (query.next()) {
            QJsonObject eventObj;
            eventObj["request_id"] = query.value("request_id").toString();
            eventObj["old_status"] = query.value("old_status").toString();
            eventObj["new_status"] = query.value("new_status").toString();
            eventObj["occurred_at"] = query.value("occurred_at").toString();
            eventsArray.append(eventObj);
        }
    } else {
        qDebug() << "[EVENT_LOGGER HATA] Olaylar cekilemedi:" << query.lastError().text();
    }

    return eventsArray;
}

void MainWindow::startServer() {
  quint16 port = settings->value("Network/Port", 1234).toUInt();
  server = new ServerHandler(port, this);

  connect(server, &ServerHandler::logReceived,      this, &MainWindow::onLogReceived);

void MainWindow::onAddDevice() {
  QString mac  = macInput->text().trimmed().toUpper();
  QString name = nameInput->text().trimmed();

  if (name.isEmpty()) {
    QMessageBox::warning(this, "Hata", "Cihaz adı boş olamaz.");
    return;
  }

  QRegularExpression macRegex("^([0-9A-F]{2}:){5}[0-9A-F]{2}$");
  if (!macRegex.match(mac).hasMatch()) {
    QMessageBox::warning(this, "Hata", "Geçersiz MAC adresi formatı!\nLütfen AA:BB:CC:DD:EE:FF formatında giriniz.");
    return;
  }

  if (dbManager.addToWhitelist(mac, name)) {
    onLogReceived(QString("[WHITELIST] Cihaz eklendi: %1  (%2)").arg(mac, name));
    macInput->clear();
    nameInput->clear();
    refreshDeviceTable();
  } else {
    QMessageBox::critical(this, "Hata", "Cihaz eklenemedi. Bu MAC zaten kayıtlı olabilir.");
  }
}

void MainWindow::onRemoveDevice() {
  int row = deviceTable->currentRow();
  if (row < 0) {
    QMessageBox::information(this, "Bilgi", "Lütfen silmek istediğiniz cihazı seçin.");
    return;
  }

  QString mac = deviceTable->item(row, 0)->text();
  auto ans = QMessageBox::question(this, "Emin misiniz?",
                                   QString("%1 MAC adresli cihazı silmek istiyor musunuz?").arg(mac));
  if (ans == QMessageBox::Yes) {
    if (dbManager.removeDevice(mac)) {
      onLogReceived(QString("[WHITELIST] Cihaz silindi: %1").arg(mac));
      refreshDeviceTable();
    }
  }
}

void MainWindow::refreshRequestTable() {
  QJsonArray reqs = dbManager.getRequests();
  requestTable->setRowCount(0);

  QMap<QString, QString> personnelMap;
  QJsonArray personnel = dbManager.getAllPersonnel();
  for (const auto &p : personnel) {
      QJsonObject pObj = p.toObject();
      personnelMap[pObj["nfc_uid"].toString()] = pObj["name"].toString();
      personnelMap[QString::number(pObj["id"].toInt())] = pObj["name"].toString();
  }

  for (const auto &val : reqs) {
    QJsonObject obj = val.toObject();
    int row = requestTable->rowCount();
    requestTable->insertRow(row);
    requestTable->setItem(row, 0, new QTableWidgetItem(obj["id"].toString()));

    QString reqId = obj["requesterId"].toString();
    QString reqName = personnelMap.value(reqId, reqId);
    requestTable->setItem(row, 1, new QTableWidgetItem(reqName));

    QString depId = obj["departmentId"].toString();
    requestTable->setItem(row, 2, new QTableWidgetItem(deptLabel(depId)));

    requestTable->setItem(row, 3, new QTableWidgetItem(obj["productId"].toString()));
    requestTable->setItem(row, 4, new QTableWidgetItem(QString::number(obj["quantity"].toInt())));

    QString status = obj["status"].toString();
    QTableWidgetItem *statusItem = new QTableWidgetItem(status);
    if (status == "TALEP_ALINDI")    statusItem->setBackground(QColor("#f39c12"));
    else if (status == "HAZIRLANIYOR") statusItem->setBackground(QColor("#3498db"));
    else if (status == "HAZIR")      statusItem->setBackground(QColor("#27ae60"));
    else if (status == "YOLDA")      statusItem->setBackground(QColor("#8e44ad"));
    else if (status == "TESLIM_EDILDI") statusItem->setBackground(QColor("#95a5a6"));
    else if (status == "IPTAL_EDILDI")  statusItem->setBackground(QColor("#e74c3c"));
    else if (status == "ESKALASYON")    statusItem->setBackground(QColor("#c0392b"));
    statusItem->setForeground(Qt::white);
    requestTable->setItem(row, 5, statusItem);

    QString rawDate = obj["createdAt"].toString();
        QDateTime dt = QDateTime::fromString(rawDate, Qt::ISODate);
        QString niceDate = dt.isValid() ? dt.toLocalTime().toString("dd.MM.yyyy HH:mm") : rawDate;
        requestTable->setItem(row, 6, new QTableWidgetItem(niceDate));
  }
}

void MainWindow::onCancelRequest() {
  int row = requestTable->currentRow();

    QMessageBox::information(this, "Bilgi", "Lütfen iptal etmek istediğiniz talebi seçin.");
    return;
  }

  QString reqId = requestTable->item(row, 0)->text();
  QString status = requestTable->item(row, 5)->text();

  if (status == "İptal Edildi" || status == "IPTAL_EDILDI" || status == "TESLIM_EDILDI" || status == "Teslim Edildi") {

       refreshRequestTable();
       QMessageBox::information(this, "Başarılı", "Talep başarıyla iptal edildi.");

       QMessageBox::warning(this, "Hata", "Talep iptal edilemedi.");

  }
}

QJsonObject WebApiHandler::processWebRequest(const QJsonObject &requestObj)
{
    QString command = requestObj["command"].toString();
    QJsonObject response;

    if (command == "GET_ALL_DASHBOARD_DATA") {
        response["status"] = "ok";
        response["devices"] = dbManager->getAllDevices();
        response["requests"] = dbManager->getRequests();
        response["inventory"] = dbManager->getAllInventoryItems();
        response["personnel"] = dbManager->getAllPersonnel();
        response["pending_cards"] = dbManager->getPendingCards();
    }
    else if (command == "GET_PENDING_CARDS") {
        response = handleGetPendingCards();
    }
    else if (command == "APPROVE_CARD") {
        response = handleApproveCard(requestObj["payload"].toObject());
    }
    else if (command == "DELETE_DEVICES") {
        QJsonArray macs = requestObj["payload"].toArray();
        for (const auto& val : macs) { dbManager->removeDevice(val.toString()); }
        response["status"] = "ok";
        response["message"] = "Seçili cihazlar başarıyla silindi.";
    }
    else if (command == "DELETE_PERSONNEL") {
        QJsonArray ids = requestObj["payload"].toArray();
        for (const auto& val : ids) { dbManager->removePersonnel(val.toString().toInt()); }
        response["status"] = "ok";
        response["message"] = "Seçili personeller başarıyla silindi.";
    }
    else if (command == "CANCEL_REQUESTS") {
        QJsonArray ids = requestObj["payload"].toArray();
        QString timeStr = QDateTime::currentDateTime().toString(Qt::ISODate);
        for (const auto& val : ids) { dbManager->cancelRequest(val.toString(), "Web üzerinden iptal edildi.", timeStr); }

void ServerHandler::broadcastMessage(const QJsonObject &messageObj) {
  QJsonDocument doc(messageObj);
  QString msgString = doc.toJson(QJsonDocument::Compact);
  for (QWebSocket *client : qAsConst(clients)) {
    if (client->isValid()) {
      client->sendTextMessage(msgString);
    }
  }
}

void ServerHandler::onNewConnection() {
  QWebSocket *clientSocket = webSocketServer->nextPendingConnection();

  connect(clientSocket, &QWebSocket::textMessageReceived, this,
          &ServerHandler::processTextMessage);
  connect(clientSocket, &QWebSocket::disconnected, this,
          &ServerHandler::socketDisconnected);

  clients << clientSocket;
  QString ip = clientSocket->peerAddress().toString();
  if (ip.startsWith("::ffff:")) {
    ip = ip.mid(7);
  }
  emitLog(QString("[BAGLANTI] Yeni cihaz baglandi. IP: %1").arg(ip));
}

void ServerHandler::processTextMessage(const QString &message) {
  QWebSocket *clientSocket = qobject_cast<QWebSocket *>(sender());
  if (!clientSocket)
    return;

  QJsonDocument doc = QJsonDocument::fromJson(message.toUtf8());

  if (doc.isNull() || !doc.isObject()) {
    clientSocket->sendTextMessage("{\"status\":\"error\", \"message\":\"Yanlis format. JSON bekleniyor.\"}");
    return;
  }

  QJsonObject jsonObj = doc.object();
  QString command = jsonObj["command"].toString();

  if (command == "GET_PENDING_CARDS" || command == "APPROVE_CARD" || command == "GET_ALL_DASHBOARD_DATA" ||
      command == "DELETE_DEVICES" || command == "DELETE_PERSONNEL" || command == "EDIT_PERSONNEL" ||
      command == "CANCEL_REQUESTS" || command == "DELETE_REQUESTS") {

      QJsonObject apiResponse = webApi->processWebRequest(jsonObj);
      QJsonDocument responseDoc(apiResponse);
      clientSocket->sendTextMessage(responseDoc.toJson(QJsonDocument::Compact));

      if (apiResponse["status"].toString() == "ok" && command != "GET_ALL_DASHBOARD_DATA" && command != "GET_PENDING_CARDS") {
          emit deviceTableChanged();
          emit requestTableChanged();
      }
      return;
  }

  // Sahadan Gelen Cihazlar İçin MAC Kontrolü
  QString rawMac = jsonObj["mac_address"].toString().trimmed();
  QString macAddress = rawMac.toUpper();

  if (macAddress.isEmpty()) {
    clientSocket->sendTextMessage(
        "{\"status\":\"error\", \"message\":\"MAC adresi bulunamadi.\"}");
    return;
  }

  else if (command == "GET_EVENTS_SINCE") {
      QString since = jsonObj["since"].toString();

      if (since.isEmpty()) {
          clientSocket->sendTextMessage("{\"status\":\"error\", \"message\":\"'since' parametresi eksik.\"}");
          return;
      }

      EventLogger eventLogger;
      QJsonArray events = eventLogger.getEventsSince(since);

      QJsonObject response;
      response["status"] = "ok";
      response["command"] = "GET_EVENTS_SINCE_RESPONSE";
      response["data"] = events;

      QJsonDocument responseDoc(response);
      clientSocket->sendTextMessage(responseDoc.toJson(QJsonDocument::Compact));

      emitLog(QString("[SYNC] %1 icin %2 tarihinden sonraki %3 olay gonderildi.")
              .arg(macAddress, since, QString::number(events.size())));
      return;
  }

  bool isAuthorized = dbManager.isMacAuthorized(macAddress);

      QString(

    clientSocket->close();
    return;
  }

  QString ipAddress = clientSocket->peerAddress().toString();
  if (ipAddress.startsWith("::ffff:")) {
    ipAddress = ipAddress.mid(7);
  }

  dbManager.updateLastSeen(macAddress, ipAddress);

  QString detectedUser = jsonObj["username"].toVariant().toString();
  if (detectedUser.isEmpty())
    detectedUser = jsonObj["requester_id"].toVariant().toString();
  if (detectedUser.isEmpty())
    detectedUser = jsonObj["user_id"].toVariant().toString();
  if (!detectedUser.isEmpty()) {

    if (!personName.isEmpty()) {
      dbManager.updateLastUser(macAddress, personName);

  }

    clientSocket->sendTextMessage(
        "{\"status\":\"error\", \"message\":\"Bilinmeyen komut.\"}");

void ServerHandler::socketDisconnected() {

bool databasemanager::createRequest(
    const QString &id, const QString &requesterId, const QString &departmentId,
    const QString &productId, int quantity, const QString &status,
    const QString &createdAt, const QString &priority) {
  QSqlQuery query;
      QSqlQuery checkStock;
      checkStock.prepare("SELECT quantity FROM inventory WHERE product_id = :pid");

                "VALUES (:id, :reqId, :depId, :prodId, :qty, :status, "
                ":priority, :createdAt)");
  query.bindValue(":id", id);
  query.bindValue(":reqId", requesterId);
  query.bindValue(":depId", departmentId);
  query.bindValue(":prodId", productId);

  query.bindValue(":priority", priority.isEmpty() ? "NORMAL" : priority);
  query.bindValue(":createdAt", createdAt);

             << "| Oncelik:" << priority;
    return true;

    qDebug() << "[DB] Envantere yeni esya eklendi:" << itemName << "| Departman:" << department;
    return true;
  }

  qDebug() << "[DB] HATA: Envantere esya eklenemedi." << query.lastError().text();
  return false;
}

bool databasemanager::checkoutInventoryItem(const QString &qrCode, const QString &username) {
  QSqlQuery checkQuery;
  checkQuery.prepare("SELECT status FROM inventory WHERE qr_code = :qr");
  checkQuery.bindValue(":qr", qrCode);
  
  if (checkQuery.exec() && checkQuery.next()) {
    QString status = checkQuery.value("status").toString();
    if (status == "KULLANIMDA") {
        return false;
    }
    
    QSqlQuery updateQuery;
    updateQuery.prepare("UPDATE inventory SET status = 'KULLANIMDA', assigned_to = :usr WHERE qr_code = :qr");
    updateQuery.bindValue(":usr", username);
    updateQuery.bindValue(":qr", qrCode);
    return updateQuery.exec();
  }
  return false;
}

bool databasemanager::checkinInventoryItem(const QString &qrCode) {
  QSqlQuery updateQuery;
  updateQuery.prepare("UPDATE inventory SET status = 'STOKTA VAR', assigned_to = '' WHERE qr_code = :qr");
  updateQuery.bindValue(":qr", qrCode);
  if (updateQuery.exec() && updateQuery.numRowsAffected() > 0) {
      return true;
  }
  return false;
}

QJsonObject databasemanager::lookupByNfcUid(const QString &nfcUid) {
  QJsonObject result;
  QSqlQuery query;
  query.prepare("SELECT id, name, department, role FROM personnel "
                "WHERE UPPER(nfc_uid) = UPPER(:nfc)");
  query.bindValue(":nfc", nfcUid.trimmed());

  if (query.exec() && query.next()) {
    result["found"] = true;
    result["id"] = query.value("id").toInt();
    result["name"] = query.value("name").toString();
    result["department"] = query.value("department").toString();
    result["role"] = query.value("role").toString();
  } else {
    result["found"] = false;
  }
  return result;
}

QString databasemanager::getPersonnelName(const QString &idOrNfc) {
  QSqlQuery query;
  query.prepare("SELECT name FROM personnel WHERE UPPER(nfc_uid) = UPPER(:val) OR id = :id_val");
  query.bindValue(":val", idOrNfc.trimmed());
  query.bindValue(":id_val", idOrNfc.toInt());
  if (query.exec() && query.next()) {
    return query.value("name").toString();
  }
  return "";
}
